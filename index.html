let rawJsonData = null; let processedResultData = null; // Otomatis ubah default Luas Plot saat Tingkat Pertumbuhan diganti document.getElementById('tingkat').addEventListener('change', function(e) { const luasInput = document.getElementById('luasPlot'); const val = e.target.value; if (val === 'Semai') luasInput.value = 4; else if (val === 'Pancang') luasInput.value = 25; else if (val === 'Tiang') luasInput.value = 100; else if (val === 'Pohon') luasInput.value = 400; }); function handleFile(e) { const file = e.target.files[0]; if (!file) return; document.getElementById('namaFile').innerText = `File terpilih: ${file.name}`; const reader = new FileReader(); reader.onload = function(evt) { const data = new Uint8Array(evt.target.result); const workbook = XLSX.read(data, { type: 'array' }); const firstSheet = workbook.Sheets[workbook.SheetNames[0]]; rawJsonData = XLSX.utils.sheet_to_json(firstSheet); const btnHitung = document.getElementById('btnHitung'); btnHitung.disabled = false; btnHitung.className = "bg-emerald-600 text-white px-6 py-2.5 rounded-lg font-bold hover:bg-emerald-700 cursor-pointer"; showStatus("✅ File berhasil dimuat! Klik tombol 'Proses Data Lapangan'.", "success"); }; reader.readAsArrayBuffer(file); } function getColumnValue(row, possibleNames) { const keys = Object.keys(row); for (let key of keys) { const cleanKey = key.toString().trim().toLowerCase(); for (let name of possibleNames) { if (cleanKey === name || cleanKey.includes(name)) { return row[key]; } } } return null; } // Format nama spesies agar rapi (Kapital Kata Pertama) function formatNamaSpesies(str) { return str.toString().trim().toLowerCase().replace(/\b\w/g, l => l.toUpperCase()); } function prosesPerhitungan() { if (!rawJsonData || rawJsonData.length === 0) { showStatus("❌ File Excel kosong atau tidak terbaca!", "error"); return; } const tingkat = document.getElementById('tingkat').value; const jumlahPlot = parseFloat(document.getElementById('jumlahPlot').value) || 0; const luasPlot = parseFloat(document.getElementById('luasPlot').value) || 0; const luasTotalHa = (jumlahPlot * luasPlot) / 10000.0; let speciesMap = {}; rawJsonData.forEach(row => { let rawNama = getColumnValue(row, ['nama spesies', 'spesies', 'jenis', 'nama jenis', 'pohon']); if (!rawNama) return; let nama = formatNamaSpesies(rawNama); let rawPlot = getColumnValue(row, ['no plot', 'plot', 'no. plot', 'nomor plot', 'id plot']) || "1"; let plot = rawPlot.toString().trim(); // Normalisasi ID plot ke String bersih let diameter = parseFloat(getColumnValue(row, ['diameter (cm)', 'diameter', 'dbh'])) || 0; let ind = parseFloat(getColumnValue(row, ['jumlah individu', 'individu', 'jumlah', 'n'])) || 1; // Hitung LBDS (m²) menggunakan standard 0.7854 atau Math.PI let dM = diameter / 100.0; let lbdsRow = (Math.PI / 4.0) * (dM * dM) * ind; if (!speciesMap[nama]) { speciesMap[nama] = { nama: nama, totalIndividu: 0, plotsSet: new Set(), totalLBDS: 0 }; } speciesMap[nama].totalIndividu += ind; speciesMap[nama].plotsSet.add(plot); speciesMap[nama].totalLBDS += lbdsRow; }); let speciesList = Object.values(speciesMap); if (speciesList.length === 0) { showStatus("❌ Kolom spesies tidak ditemukan di file Excel!", "error"); return; } // 1. Hitung Nilai Absolut & Total Komunitas let totalK = 0, totalF = 0, totalD = 0; speciesList.forEach(sp => { sp.k = luasTotalHa > 0 ? sp.totalIndividu / luasTotalHa : 0; sp.f = jumlahPlot > 0 ? sp.plotsSet.size / jumlahPlot : 0; sp.dom = luasTotalHa > 0 ? sp.totalLBDS / luasTotalHa : 0; totalK += sp.k; totalF += sp.f; totalD += sp.dom; }); // 2. Hitung Nilai Relatif & INP processedResultData = speciesList.map(sp => { let kr = totalK > 0 ? (sp.k / totalK) * 100 : 0; let fr = totalF > 0 ? (sp.f / totalF) * 100 : 0; let isSemaiPancang = (tingkat === 'Semai' || tingkat === 'Pancang'); let dr = isSemaiPancang ? 0 : (totalD > 0 ? (sp.dom / totalD) * 100 : 0); let inp = kr + fr + dr; return { "Nama Spesies": sp.nama, "Total Individu": sp.totalIndividu, "Plot Ditemukan": sp.plotsSet.size, "Kerapatan (K)": Number(sp.k.toFixed(2)), "KR (%)": Number(kr.toFixed(2)), "Frekuensi (F)": Number(sp.f.toFixed(2)), "FR (%)": Number(fr.toFixed(2)), "Dominansi (D)": isSemaiPancang ? 0 : Number(sp.dom.toFixed(2)), "DR (%)": isSemaiPancang ? 0 : Number(dr.toFixed(2)), "INP (%)": Number(inp.toFixed(2)) }; }); renderTable(); const btnDownload = document.getElementById('btnDownload'); btnDownload.disabled = false; btnDownload.className = "bg-blue-600 text-white px-6 py-2.5 rounded-lg font-bold hover:bg-blue-700 cursor-pointer"; showStatus("✅ Perhitungan selesai! Hasil rekap dan INP telah diperbarui secara presisi.", "success"); } function renderTable() { const tbody = document.getElementById('tabelHasilBody'); tbody.innerHTML = ''; processedResultData.forEach(row => { const tr = document.createElement('tr'); tr.className = 'hover:bg-slate-50'; tr.innerHTML = ` ${row["Nama Spesies"]} ${row["Total Individu"]} ${row["Plot Ditemukan"]} ${row["Kerapatan (K)"]} ${row["KR (%)"]}% ${row["Frekuensi (F)"]} ${row["FR (%)"]}% ${row["Dominansi (D)"]} ${row["DR (%)"]}% ${row["INP (%)"]}% `; tbody.appendChild(tr); }); document.getElementById('areaHasil').classList.remove('hidden'); } function downloadExcel() { if (!processedResultData) return; const worksheet = XLSX.utils.json_to_sheet(processedResultData); const workbook = XLSX.utils.book_new(); XLSX.utils.book_append_sheet(workbook, worksheet, "Hasil Rekap INP"); XLSX.writeFile(workbook, "Hasil_Rekap_INP_Vegetasi.xlsx"); } function showStatus(msg, type) { const box = document.getElementById('statusBox'); box.innerText = msg; box.classList.remove('hidden', 'bg-emerald-100', 'text-emerald-800', 'bg-red-100', 'text-red-800'); if (type === 'success') { box.classList.add('bg-emerald-100', 'text-emerald-800'); } else { box.classList.add('bg-red-100', 'text-red-800'); } }
